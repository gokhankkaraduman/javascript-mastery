## 1. Object Methods

**Methods** are functions stored as object properties. They allow objects to have behaviors, not just data.

### A. Basic Object Methods
```javascript
let person = {
    name: "John",
    age: 30,
    
    // Method using function keyword
    greet: function() {
        console.log("Hello!");
    },
    
    // Method shorthand (ES6)
    introduce() {
        console.log("My name is John");
    },
    
    // Method with parameters
    sayHello(name) {
        console.log("Hello, " + name + "!");
    }
};

// Calling methods
person.greet();           // "Hello!"
person.introduce();       // "My name is John"
person.sayHello("Alice"); // "Hello, Alice!"
```

---

### B. Methods That Return Values
```javascript
let calculator = {
    add(a, b) {
        return a + b;
    },
    
    subtract(a, b) {
        return a - b;
    },
    
    multiply(a, b) {
        return a * b;
    },
    
    divide(a, b) {
        if (b === 0) {
            return "Cannot divide by zero";
        }
        return a / b;
    }
};

console.log(calculator.add(5, 3));       // 8
console.log(calculator.multiply(4, 5));  // 20
console.log(calculator.divide(10, 2));   // 5
```

---

### C. Methods Accessing Other Properties
```javascript
let product = {
    name: "Laptop",
    price: 999,
    quantity: 5,
    
    // Method accessing other properties
    getTotalValue() {
        return this.price * this.quantity;
    },
    
    getInfo() {
        return this.name + " - $" + this.price;
    },
    
    applyDiscount(percentage) {
        this.price = this.price * (1 - percentage / 100);
        return this.price;
    }
};

console.log(product.getTotalValue());    // 4995
console.log(product.getInfo());          // "Laptop - $999"
product.applyDiscount(10);
console.log(product.price);              // 899.1
```

## 2. The `this` Keyword

**`this`** refers to the object that is executing the current function.

### A. `this` in Object Methods
```javascript
let user = {
    firstName: "John",
    lastName: "Doe",
    age: 30,
    
    getFullName() {
        return this.firstName + " " + this.lastName;
    },
    
    introduce() {
        return "Hi, I'm " + this.firstName + " and I'm " + this.age + " years old";
    },
    
    haveBirthday() {
        this.age++;
        return "Happy birthday! You are now " + this.age;
    }
};

console.log(user.getFullName());    // "John Doe"
console.log(user.introduce());      // "Hi, I'm John and I'm 30 years old"
console.log(user.haveBirthday());   // "Happy birthday! You are now 31"
```

**Why use `this`?**
- Access other properties in the same object
- Make methods reusable
- Modify object's own properties

---

### B. `this` Without Arrow Functions
```javascript
let counter = {
    count: 0,
    
    increment() {
        this.count++;
        return this.count;
    },
    
    decrement() {
        this.count--;
        return this.count;
    },
    
    reset() {
        this.count = 0;
        return this.count;
    },
    
    getValue() {
        return this.count;
    }
};

console.log(counter.increment());  // 1
console.log(counter.increment());  // 2
console.log(counter.increment());  // 3
console.log(counter.getValue());   // 3
console.log(counter.reset());      // 0
```

---

### C. Common Pitfall: Arrow Functions and `this`

⚠️ **Arrow functions don't have their own `this`!**
```javascript
let person = {
    name: "Alice",
    
    // ✅ Regular function - 'this' works
    greet: function() {
        console.log("Hello, I'm " + this.name);
    },
    
    // ❌ Arrow function - 'this' doesn't refer to person
    greetArrow: () => {
        console.log("Hello, I'm " + this.name);  // undefined
    }
};

person.greet();       // "Hello, I'm Alice" ✅
person.greetArrow();  // "Hello, I'm undefined" ❌
```

**Rule:** Use regular functions for object methods when you need `this`.

### D. Practical Examples with `this`

#### Bank Account
```javascript
let bankAccount = {
    owner: "John Doe",
    balance: 1000,
    
    deposit(amount) {
        this.balance += amount;
        return `Deposited $${amount}. New balance: $${this.balance}`;

```

#### Shopping Cart
```javascript
let cart = {
```

## 3. Spread Operator (`...`)

The **spread operator** expands iterables (arrays, objects) into individual elements.

### A. Spread with Arrays

#### Copying Arrays
```javascript
let original = [1, 2, 3, 4, 5];
let copy = [...original];

copy.push(6);

console.log(original);  // [1, 2, 3, 4, 5]
console.log(copy);      // [1, 2, 3, 4, 5, 6]
```

#### Combining Arrays
```javascript
let fruits = ["apple", "banana"];
let vegetables = ["carrot", "potato"];

let food = [...fruits, ...vegetables];
console.log(food);  // ["apple", "banana", "carrot", "potato"]

// Add elements while spreading
let allFood = ["bread", ...fruits, "cheese", ...vegetables];
console.log(allFood);
// ["bread", "apple", "banana", "cheese", "carrot", "potato"]
```

#### Spreading in Function Calls
```javascript
let numbers = [5, 12, 8, 20, 3];

// Without spread (wrong)
console.log(Math.max(numbers));  // NaN

// With spread (correct)
console.log(Math.max(...numbers));  // 20
console.log(Math.min(...numbers));  // 3

// Example with custom function
function sum(a, b, c) {
    return a + b + c;
}

let nums = [10, 20, 30];
console.log(sum(...nums));  // 60
```

### B. Spread with Objects

#### Copying Objects
```javascript
let person = {
    name: "John",
    age: 30,
    city: "New York"
};

let personCopy = {...person};
personCopy.name = "Jane";

console.log(person.name);      // "John" (original unchanged)
console.log(personCopy.name);  // "Jane"
```

#### Merging Objects
```javascript
let basicInfo = {
    name: "Alice",
    age: 25
};

let contactInfo = {
    email: "alice@example.com",
    phone: "555-0100"
};

let completeProfile = {...basicInfo, ...contactInfo};
console.log(completeProfile);
// {name: "Alice", age: 25, email: "alice@example.com", phone: "555-0100"}
```

#### Overriding Properties
```javascript
let defaults = {
    theme: "light",
    language: "en",
    notifications: true
};

let userPreferences = {
    theme: "dark",
    language: "es"
};

// Later spreads override earlier ones
let settings = {...defaults, ...userPreferences};
console.log(settings);
// {theme: "dark", language: "es", notifications: true}
```

#### Adding/Updating Properties
```javascript
let user = {
    name: "Bob",
    age: 28
};

// Add new property
let updatedUser = {...user, city: "Chicago"};
console.log(updatedUser);
// {name: "Bob", age: 28, city: "Chicago"}

// Update existing property
let olderUser = {...user, age: 29};
console.log(olderUser);
// {name: "Bob", age: 29}
```

### C. Spread in Practical Scenarios

#### Removing Properties
```javascript
let user = {
    id: 1,
    username: "john_doe",
    password: "secret123",
    email: "john@example.com"
};

// Remove password by destructuring and spreading rest
let {password, ...safeUser} = user;
console.log(safeUser);
// {id: 1, username: "john_doe", email: "john@example.com"}
```

#### Updating Nested Objects
```javascript
let profile = {
    name: "Alice",
    settings: {
        theme: "light",
        notifications: true
    }
};

// Update nested property
let updatedProfile = {
    ...profile,
    settings: {
        ...profile.settings,
        theme: "dark"
    }
};

console.log(updatedProfile.settings);
// {theme: "dark", notifications: true}
```

## 4. Rest Parameters (`...`)

**Rest parameters** collect multiple arguments into an array. Different use than spread!

### A. Basic Rest Parameters
```javascript
// Traditional way - limited parameters
function oldSum(a, b, c) {
    return a + b + c;
}

// With rest - unlimited parameters
function sum(...numbers) {
    let total = 0;
    for (let num of numbers) {
        total += num;
    }
    return total;
}

console.log(sum(1, 2, 3));           // 6
console.log(sum(1, 2, 3, 4, 5));     // 15
console.log(sum(10, 20, 30, 40, 50)); // 150
```

---

### B. Rest with Other Parameters

Rest parameter must be **last**!
```javascript
// ✅ Correct
function greet(greeting, ...names) {
    for (let name of names) {
        console.log(greeting + ", " + name);
    }
}

greet("Hello", "Alice", "Bob", "Charlie");
// Hello, Alice
// Hello, Bob
// Hello, Charlie

// ❌ Wrong - rest must be last
// function wrong(...items, last) { }
```

---

### C. Rest vs Arguments
```javascript
// Old way - arguments object
function oldWay() {
    console.log(arguments);  // Array-like, not real array
    // arguments.map()  // ❌ Doesn't work
}

// Modern way - rest parameters
function modernWay(...args) {
    console.log(args);  // Real array
    args.map(x => x * 2);  // ✅ Works!
}

modernWay(1, 2, 3);  // [1, 2, 3]
```

### D. Practical Examples with Rest

#### Find Maximum
```javascript
function findMax(...numbers) {
    if (numbers.length === 0) return undefined;
    
    let max = numbers[0];
    for (let num of numbers) {
        if (num > max) {
            max = num;
        }
    }
    return max;
}

console.log(findMax(5, 12, 8, 20, 3));  // 20
console.log(findMax(100, 50, 75));      // 100
```

#### Filter by Type
```javascript
function numbersOnly(...items) {
    let result = [];
    for (let item of items) {
        if (typeof item === "number") {
            result.push(item);
        }
    }
    return result;
}

console.log(numbersOnly(1, "a", 2, "b", 3, true, 4));
// [1, 2, 3, 4]
```

#### Multiple All Arguments
```javascript
function multiply(multiplier, ...numbers) {
    return numbers.map(num => num * multiplier);
}

console.log(multiply(2, 1, 2, 3, 4, 5));
// [2, 4, 6, 8, 10]
```

## 6. Combining Everything

### Example: Todo List Manager
```javascript
let todoList = {
    todos: [],
    
    // Add todo using rest parameters
    addTodos(...items) {
        for (let item of items) {
            this.todos.push({
                text: item,
                completed: false,
                id: Date.now() + Math.random()
            });
        }
        return `Added ${items.length} todo(s)`;
    },
    
    // Get all todos using spread
    getAllTodos() {
        return [...this.todos];  // Return copy
    },
    
    // Complete a todo
    completeTodo(id) {
        for (let todo of this.todos) {
            if (todo.id === id) {
                todo.completed = true;
                return `Completed: ${todo.text}`;
            }
        }
        return "Todo not found";
    },
    
    // Get uncompleted todos
    getActiveTodos() {
        let active = [];
        for (let todo of this.todos) {
            if (!todo.completed) {
                active.push({...todo});  // Use spread to copy
            }
        }
        return active;
    }
};

todoList.addTodos("Learn JavaScript", "Build project", "Practice coding");
console.log(todoList.getAllTodos());
```

### Example: Data Merger
```javascript
let dataMerger = {
    // Merge multiple objects
    mergeObjects(...objects) {
        let result = {};
        for (let obj of objects) {
            result = {...result, ...obj};
        }
        return result;
    },
    
    // Merge arrays and remove duplicates
    mergeArrays(...arrays) {
        let combined = [];
        for (let arr of arrays) {
            combined = [...combined, ...arr];
        }
        // Remove duplicates
        return [...new Set(combined)];
    }
};

let obj1 = {a: 1, b: 2};
let obj2 = {c: 3, d: 4};
let obj3 = {e: 5};

console.log(dataMerger.mergeObjects(obj1, obj2, obj3));
// {a: 1, b: 2, c: 3, d: 4, e: 5}

let arr1 = [1, 2, 3];
let arr2 = [3, 4, 5];
let arr3 = [5, 6, 7];

console.log(dataMerger.mergeArrays(arr1, arr2, arr3));
// [1, 2, 3, 4, 5, 6, 7]
```

## 💡 Key Takeaways

### ✅ Object Methods
- Functions stored as object properties
- Use regular functions (not arrow) when you need `this`
- Access other properties with `this.propertyName`
- Make objects interactive and reusable

### ✅ `this` Keyword
- Refers to the object executing the method
- Use to access/modify object's own properties
- **Don't use arrow functions** for methods that need `this`

### ✅ Spread Operator (`...`)
- **Arrays**: Copy, combine, pass to functions
- **Objects**: Copy, merge, add/update properties
- Creates **shallow copies**
- Syntax: `[...array]` or `{...object}`

### ✅ Rest Parameters (`...`)
- Collects multiple arguments into an array
- Must be the **last parameter**
- Creates a **real array** (unlike `arguments`)
- Syntax: `function(...args) { }`

---

## 📊 Spread vs Rest

| Feature | Spread | Rest |
|---------|--------|------|
| **Purpose** | Expand/unpack | Collect/pack |
| **Where** | Array/object literals, function calls | Function parameters, destructuring |
| **Result** | Individual elements | Single array |
| **Example** | `[...arr]`, `{...obj}` | `function(...args)` |
```javascript
// SPREAD - Expands
let arr = [1, 2, 3];
console.log(...arr);  // 1 2 3 (individual values)

// REST - Collects
function sum(...numbers) {
    return numbers;  // [1, 2, 3] (array)
}
console.log(sum(1, 2, 3));
```

## 🎯 Common Patterns

### Pattern 1: Clone and Modify
```javascript

```

### Pattern 2: Merge with Defaults
```javascript
let defaults = {theme: "light", notifications: true};
let userSettings = {theme: "dark"};
let final = {...defaults, ...userSettings};
```

### Pattern 3: Flexible Function Arguments
```javascript
function calculate(operation, ...numbers) {
    // Handle any number of arguments
}
```

### Pattern 4: Remove Property
```javascript
let {unwanted, ...keep} = object;
```
