## 1. Advanced Array Iteration

### A. `for...of` Loop (ES6)

The modern, cleaner way to iterate through array **values**.

**Syntax:**
```javascript
for (let element of array) {
    // code to execute for each element
}
```

**Examples:**
```javascript
let fruits = ["apple", "banana", "orange"];

for (let fruit of fruits) {
    console.log(fruit);
}
// Output:
// apple
// banana
// orange

// With numbers
let numbers = [10, 20, 30, 40, 50];
for (let num of numbers) {
    console.log(num * 2);
}
// Output: 20, 40, 60, 80, 100

// With strings (iterates characters)
let word = "Hello";
for (let char of word) {
    console.log(char);
}
// Output: H, e, l, l, o
```

**When to use:**
- When you need the **values** (not indices)
- Clean, readable code
- Works with any iterable (arrays, strings, etc.)

---

### B. Traditional `for` Loop with Index

When you need both **index** and **value**.
```javascript
let colors = ["red", "green", "blue"];

for (let i = 0; i < colors.length; i++) {
    console.log(`Index ${i}: ${colors[i]}`);
}
// Output:
// Index 0: red
// Index 1: green
// Index 2: blue
```

---

### C. `for...in` Loop (For Objects, Not Recommended for Arrays)

Iterates over **keys/indices** (not values).
```javascript
let fruits = ["apple", "banana", "orange"];

for (let index in fruits) {
    console.log(index, fruits[index]);
}
// Output:
// 0 apple
// 1 banana
// 2 orange

// ⚠️ Better to use for...of for arrays
```

## 2. Scope: Where Variables Live

**Scope** determines where variables can be accessed in your code.

### A. Global Scope

Variables declared **outside** any function are in the **global scope**.
```javascript
let globalVar = "I'm global";

function showGlobal() {
    console.log(globalVar);  // Can access global variable
}

showGlobal();  // "I'm global"
console.log(globalVar);  // "I'm global" (accessible everywhere)
```

---

### B. Function Scope (Local Scope)

Variables declared **inside** a function are in the **function scope**.
```javascript
function myFunction() {
    let localVar = "I'm local";
    console.log(localVar);  // Works inside function
}

myFunction();  // "I'm local"
console.log(localVar);  // ❌ ERROR: localVar is not defined
```

**Key Rule:** Variables inside functions **cannot** be accessed from outside.

---

### C. Block Scope (`let` and `const`)

Variables declared with `let` or `const` inside `{}` are **block-scoped**.
```javascript
if (true) {
    let blockVar = "I'm in a block";
    console.log(blockVar);  // Works inside block
}

console.log(blockVar);  // ❌ ERROR: blockVar is not defined

// Same with loops
for (let i = 0; i < 3; i++) {
    let loopVar = "Inside loop";
}
console.log(loopVar);  // ❌ ERROR
console.log(i);  // ❌ ERROR
```

---

### D. Scope Chain

Inner functions can access variables from outer functions.
```javascript
let global = "Global";

function outer() {
    let outerVar = "Outer";
    
    function inner() {
        let innerVar = "Inner";
        console.log(global);     // ✅ Can access
        console.log(outerVar);   // ✅ Can access
        console.log(innerVar);   // ✅ Can access
    }
    
    inner();
    console.log(innerVar);  // ❌ ERROR: Can't access inner variable
}

outer();
```

**Scope Hierarchy:**
```
Global Scope
  └── Outer Function Scope
        └── Inner Function Scope
```

---

### E. `var` vs `let` vs `const` (Scope Differences)
```javascript
// var is function-scoped
function testVar() {
    if (true) {
        var x = 10;
    }
    console.log(x);  // ✅ 10 (var ignores block scope)
}

// let is block-scoped
function testLet() {
    if (true) {
        let y = 10;
    }
    console.log(y);  // ❌ ERROR (let respects block scope)
}

// const is also block-scoped
function testConst() {
    if (true) {
        const z = 10;
    }
    console.log(z);  // ❌ ERROR (const respects block scope)
}
```

**Best Practice:** Use `let` and `const` (not `var`).

## 3. The `arguments` Object

Inside every function, there's a special **array-like** object called `arguments` that contains all arguments passed to the function.

### What is `arguments`?
```javascript
function showArguments() {
    console.log(arguments);
}

showArguments(1, 2, 3, 4, 5);
// Output: [1, 2, 3, 4, 5] (array-like object)
```

### Key Characteristics:

- **Array-like** but NOT a real array
- Has `length` property
- Can access elements by index: `arguments[0]`, `arguments[1]`, etc.
- Does NOT have array methods like `push()`, `pop()`, etc.

---

### Using `arguments` Without Parameters
```javascript
// No parameters defined, but can still receive arguments
function sum() {
    let total = 0;
    
    for (let i = 0; i < arguments.length; i++) {
        total += arguments[i];
    }
    
    return total;
}

console.log(sum(1, 2, 3));        // 6
console.log(sum(10, 20, 30, 40)); // 100
console.log(sum(5));              // 5
```

---

### Flexible Number of Arguments
```javascript
function multiply() {
    let result = 1;
    
    for (let i = 0; i < arguments.length; i++) {
        result *= arguments[i];
    }
    
    return result;
}

console.log(multiply(2, 3));           // 6
console.log(multiply(2, 3, 4));        // 24
console.log(multiply(1, 2, 3, 4, 5));  // 120
```

---

### Converting `arguments` to Real Array
```javascript
function myFunction() {
    // Method 1: Array.from()
    let argsArray1 = Array.from(arguments);
    
    // Method 2: Spread operator
    let argsArray2 = [...arguments];
    
    // Method 3: slice (old way)
    let argsArray3 = Array.prototype.slice.call(arguments);
    
    console.log(argsArray1);  // Real array now!
    argsArray1.push(100);     // Array methods work
}

myFunction(1, 2, 3);
```

---

### `arguments` with Named Parameters
```javascript
function greet(greeting, name) {
    console.log(greeting + ", " + name);
    console.log("All arguments:", arguments);
}

greet("Hello", "John", "extra", "arguments");
// Output:
// Hello, John
// All arguments: ["Hello", "John", "extra", "arguments"]
```

---

### Finding Max in `arguments`
```javascript
function findMax() {
    if (arguments.length === 0) {
        return undefined;
    }
    
    let max = arguments[0];
    
    for (let i = 1; i < arguments.length; i++) {
        if (arguments[i] > max) {
            max = arguments[i];
        }
    }
    
    return max;
}

console.log(findMax(5, 12, 8, 20, 3));  // 20
console.log(findMax(100, 50, 75));      // 100
```

---

### Searching in `arguments`
```javascript
function contains(searchValue) {
    for (let i = 0; i < arguments.length; i++) {
        if (arguments[i] === searchValue) {
            return true;
        }
    }
    return false;
}

console.log(contains(3, 1, 2, 3, 4, 5));  // true
console.log(contains(10, 1, 2, 3, 4, 5)); // false
```

---

### ⚠️ Important Notes About `arguments`

1. **Not available in arrow functions:**
```javascript
const arrow = () => {
    console.log(arguments);  // ❌ ERROR: arguments is not defined
};
```

2. **Array-like, not a real array:**
```javascript
function test() {
    arguments.push(5);  // ❌ ERROR: push is not a function
    
    // Convert to array first:
    let args = [...arguments];
    args.push(5);  // ✅ Works
}
```

## 4. Hoisting

**Hoisting** is JavaScript's behavior of moving declarations to the top of their scope before code execution.

### A. Variable Hoisting

#### `var` Hoisting
```javascript
console.log(x);  // undefined (not an error!)
var x = 5;
console.log(x);  // 5

// What actually happens:
var x;  // Declaration hoisted
console.log(x);  // undefined
x = 5;  // Assignment stays in place
console.log(x);  // 5
```

#### `let` and `const` Hoisting
```javascript
console.log(y);  // ❌ ReferenceError: Cannot access 'y' before initialization
let y = 10;

console.log(z);  // ❌ ReferenceError: Cannot access 'z' before initialization
const z = 20;
```

**Temporal Dead Zone (TDZ):** `let` and `const` are hoisted but can't be accessed before declaration.

---

### B. Function Hoisting

#### Function Declarations (Fully Hoisted)
```javascript
greet();  // ✅ Works! "Hello"

function greet() {
    console.log("Hello");
}
```

The entire function is hoisted to the top.

#### Function Expressions (NOT Hoisted)
```javascript
sayHi();  // ❌ ERROR: sayHi is not a function

var sayHi = function() {
    console.log("Hi");
};

// What actually happens:
var sayHi;  // Variable hoisted as undefined
sayHi();    // Trying to call undefined as function
sayHi = function() { ... };  // Assignment happens here
```

---

### C. Hoisting Order

1. Function declarations
2. Variable declarations
```javascript
console.log(typeof myFunc);  // "function"
console.log(typeof myVar);   // "undefined"

var myVar = "Hello";

function myFunc() {
    return "Function";
}
```

---

### D. Best Practices to Avoid Hoisting Issues

1. **Always declare variables at the top of their scope**
```javascript
function myFunction() {
    let x = 10;  // Declare at top
    let y = 20;
    // Rest of code
}
```

2. **Use `let` and `const` instead of `var`**
```javascript
// ❌ Avoid
var x = 10;

// ✅ Use
let x = 10;
const y = 20;
```

3. **Define functions before using them**
```javascript
// ✅ Good practice
function greet() {
    console.log("Hello");
}

greet();
```

## 💡 Key Takeaways

### ✅ Array Iteration
- **`for...of`**: Iterate through values (modern, clean)
- **`for` loop**: When you need index and value
- **`for...in`**: For objects (not recommended for arrays)

### ✅ Scope
- **Global Scope**: Variables accessible everywhere
- **Function Scope**: Variables only accessible inside function
- **Block Scope**: Variables with `let`/`const` inside `{}`
- **Inner functions can access outer variables, but NOT vice versa**

### ✅ `arguments` Object
- Special array-like object in functions
- Contains all passed arguments
- Has `length` property
- Can be converted to real array with `[...arguments]` or `Array.from()`
- **Not available in arrow functions**

### ✅ Hoisting
- **`var`**: Declaration hoisted, initialized as `undefined`
- **`let`/`const`**: Hoisted but in Temporal Dead Zone (can't access)
- **Function declarations**: Fully hoisted
- **Function expressions**: Only variable declaration hoisted

## 🎯 Common Patterns

### Pattern 1: Sum All Arguments
```javascript
function sum() {
    let total = 0;
    for (let i = 0; i < arguments.length; i++) {
        total += arguments[i];
    }
    return total;
}
```

### Pattern 2: Safe Variable Access
```javascript
function outer() {
    let outerVar = "Accessible";
    
    function inner() {
        console.log(outerVar);  // ✅ Can access
    }
    
    return inner;
}
```

### Pattern 3: Avoid Hoisting Issues
```javascript
// ✅ Declare at top
function myFunction() {
    let a = 10;
    let b = 20;
    
    // Use variables
    console.log(a + b);
}
```
