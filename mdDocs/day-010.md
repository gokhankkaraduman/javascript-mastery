## 💡 Key Concepts to Master

### Variables & Data Types
```javascript
// Variable declaration
let name = "John";
const age = 25;
var city = "New York";

// Data types
typeof "hello"    // "string"
typeof 42         // "number"
typeof true       // "boolean"
typeof undefined  // "undefined"
typeof null       // "object" (historical bug)
```

### Functions
```javascript
// Regular function
function add(a, b) {
    return a + b;
}

// Arrow function
const multiply = (a, b) => a * b;

// Callback
function calculate(a, b, operation) {
    return operation(a, b);
}
```

### Arrays
```javascript
let arr = [1, 2, 3, 4, 5];

// Methods
arr.push(6);           // Add to end
arr.pop();             // Remove from end
arr.map(x => x * 2);   // Transform
arr.forEach(x => console.log(x));  // Iterate
```

### Objects
```javascript
let person = {
    name: "Alice",
    age: 25,
    greet() {
        console.log("Hello, " + this.name);
    }
};

// Access
person.name
person["age"]

// Loop
for (let key in person) {
    console.log(key, person[key]);
}
```

### Conditionals
```javascript
if (condition) {
    // code
} else if (otherCondition) {
    // code
} else {
    // code
}

// Ternary
let result = condition ? "yes" : "no";
```

### Loops
```javascript
// For loop
for (let i = 0; i < 5; i++) {
    console.log(i);
}

// For...of
for (let item of array) {
    console.log(item);
}

// For...in
for (let key in object) {
    console.log(key);
}
```

## 🎓 Common Patterns to Practice

### Pattern 1: Array Transformation
```javascript
// Transform data from one format to another
let numbers = [1, 2, 3, 4, 5];
let doubled = numbers.map(n => n * 2);
```

### Pattern 2: Filtering Data
```javascript
// Extract items that meet criteria
let numbers = [1, 2, 3, 4, 5, 6];
let evens = [];
numbers.forEach(n => {
    if (n % 2 === 0) evens.push(n);
});
```

### Pattern 3: Aggregation
```javascript
// Calculate sum, average, max, min
let numbers = [10, 20, 30, 40];
let sum = 0;
numbers.forEach(n => sum += n);
let average = sum / numbers.length;
```

### Pattern 4: Object Manipulation
```javascript
// Create, update, transform objects
let users = [
    {name: "Alice", age: 25},
    {name: "Bob", age: 30}
];

let names = users.map(u => u.name);
```

### Pattern 5: Nested Structures
```javascript
// Work with complex data
let company = {
    name: "Tech Corp",
    employees: [
        {name: "Alice", role: "Developer"},
        {name: "Bob", role: "Designer"}
    ]
};
```
